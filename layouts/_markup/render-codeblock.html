{{- /* Code blocks — Markdown render hook for fenced code blocks (site override). Opt-in extras, written in the info string: {title="index.html"} → header bar with a title or a file name {icon=true} → icon of the code language {icon="python"} → explicit icon (Simple Icons name, "ui:x" for Lucide) {copy=true} → copy-to-clipboard button (icon only, labelled via aria-label) Site-wide defaults (config/_default/params.yaml), the info string always wins. A block that asks for none of these goes through transform.HighlightCodeBlock, Hugo’s own default rendering. The figure is auto-synced with the Chroma style: in inline mode (noClasses, the Hugo default) Chroma writes the theme background — and its default text color, when the Background entry defines one — inline on the 
. The
  hook extracts those colors and injects them as CSS custom properties on the
  figure, so figure and figcaption follow any markup.highlight.style change
  automatically. The static tokens in assets/css/site.css are only the
  fallback if this extraction fails.

  @see layouts/_partials/func/GetCodeIcon.html for the language → icon mapping.
*/ -}}
{{- $title := "" -}}
{{- $icon := "" -}}
{{- $copy := false -}}
{{- $autoIcon := false -}}
{{- $siteIcon := false -}}
{{- $siteCopy := false -}}
{{- with site.Params.code }}
  {{- with .icon }}{{- $siteIcon = true -}}{{- end }}
  {{- with .copy }}{{- $siteCopy = true -}}{{- end }}
{{- end }}
{{- with .Attributes }}
  {{- with .title }}{{- $title = printf "%v" . -}}{{- end }}
  {{- if isset . "icon" }}
    {{- $value := printf "%v" (index . "icon") -}}
    {{- if eq $value "true" }}
      {{- $autoIcon = true -}}
    {{- else if eq $value "false" }}
      {{- /* explicit opt-out */ -}}
    {{- else }}
      {{- $icon = $value -}}
    {{- end }}
  {{- else if $siteIcon }}{{- $autoIcon = true -}}{{- end }}
  {{- if isset . "copy" }}
    {{- if eq (printf "%v" (index . "copy")) "true" }}{{- $copy = true -}}{{- end }}
  {{- else if $siteCopy }}{{- $copy = true -}}{{- end }}
{{- else }}
  {{- if $siteIcon }}{{- $autoIcon = true -}}{{- end }}
  {{- if $siteCopy }}{{- $copy = true -}}{{- end }}
{{- end }}
{{- if $autoIcon }}{{- $icon = partial "func/GetCodeIcon.html" .Type -}}{{- end }}
{{- if not (or $title $icon $copy) -}}
  {{- transform.HighlightCodeBlock . -}}
{{- else }}
  {{- $attrs := slice -}}
  {{- range $key, $value := $.Attributes }}
    {{- if not (in (slice "title" "icon" "copy") $key) }}
      {{- $attrs = $attrs | append (printf `%s="%s"` $key (htmlEscape (printf "%v" $value))) -}}
    {{- end }}
  {{- end }}
  {{- $code := highlight $.Inner $.Type $.Options -}}
  {{- /* Auto-sync with the Chroma style: extract the colors Chroma writes
         inline on the 
 (3-to-8-digit hex, Chroma may compress #ffffff
         to #fff) and cascade them to the figure via CSS custom properties. */ -}}
  {{- $preTag := "" -}}
  {{- with findRE `]*>` $code 1 }}{{- $preTag = index . 0 -}}{{- end }}
  {{- $style := "" -}}
  {{- with findRE `background-color: ?#[0-9a-fA-F]{3,8}` $preTag 1 }}
    {{- $style = printf `%s--code-block-color-background: %s;` $style (index . 0 | replaceRE `.*:` ``) -}}
  {{- end }}
  {{- /* default text color: "color:" not preceded by "-", so background-color
         is never mistaken for the text color */ -}}
  {{- with findRE `(^|[^-])color: ?#[0-9a-fA-F]{3,8}` $preTag 1 }}
    {{- $style = printf `%s--code-block-color-header-text: %s;` $style (index . 0 | replaceRE `.*:` ``) -}}
  {{- end }}
  
{{- end -}}
